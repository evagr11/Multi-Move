150 void FixedUpdate()
151 {
152 if (isOnGround)
153 {
154 // Si la velocidad horizontal está dentro del límite, aumenta la velocidad con la aceleración del suelo
155 if (rb.velocity.x < stats.maxGroundHorizontalSpeed && rb.velocity.x > -stats.maxGroundHorizontalSpeed)
156 {
157 rb.velocity = new Vector2(rb.velocity.x + (direction.x * stats.groundAcceleration * Time.deltaTime), rb.velocity.y);
158 }
159 else
160 {
161 // Si supera la velocidad máxima, se restringe al máximo permitido
162 if (rb.velocity.x > stats.maxGroundHorizontalSpeed)
163 rb.velocity = new Vector2(stats.maxGroundHorizontalSpeed, rb.velocity.y);
164 else if (rb.velocity.x < -stats.maxGroundHorizontalSpeed)
165 rb.velocity = new Vector2(-stats.maxGroundHorizontalSpeed, rb.velocity.y);
166 }
167 }
168 else
169 {
170 // Si la velocidad horizontal está dentro del límite, aplica la aceleración para moverse en el aire
171 if (rb.velocity.x < stats.maxAirHorizontalSpeed && rb.velocity.x > -stats.maxAirHorizontalSpeed)
172 {
173 rb.velocity = new Vector2(rb.velocity.x + (direction.x * stats.airAcceleration * Time.deltaTime), rb.velocity.y);
174 }
175 else
176 {
177 // Si la velocidad supera el máximo, se restringe
178 if (rb.velocity.x > stats.maxAirHorizontalSpeed)
179 rb.velocity = new Vector2(stats.maxAirHorizontalSpeed, rb.velocity.y);
180 else if (rb.velocity.x < -stats.maxAirHorizontalSpeed)
181 rb.velocity = new Vector2(-stats.maxAirHorizontalSpeed, rb.velocity.y);
182 }
183 }
184
185 if (isJumping)
186 {
187 rb.velocity = new Vector2(rb.velocity.x, stats.jumpStrength);
188 isJumping = false;
189 }
190
191 if (isOnGround)
192 {
193 // Si el personaje se mueve a la derecha, reduce la velocidad aplicando la fricción del suelo
194 if (rb.velocity.x > 0)
195 rb.velocity = new Vector2(rb.velocity.x - stats.groundHorizontalFriction * Time.deltaTime, rb.velocity.y);
196 // Idem para la izquierda
197 else if (rb.velocity.x < 0)
198 rb.velocity = new Vector2(rb.velocity.x + stats.groundHorizontalFriction * Time.deltaTime, rb.velocity.y);
199 }
200 else //mismo mecanismo para cuando esta en el aire
201 {
202 if (rb.velocity.x > 0)
203 rb.velocity = new Vector2(rb.velocity.x - stats.airFriccion * Time.deltaTime, rb.velocity.y);
204 else if (rb.velocity.x < 0)
205 rb.velocity = new Vector2(rb.velocity.x + stats.airFriccion * Time.deltaTime, rb.velocity.y);
206 }
207
208 if (rb.velocity.y < -stats.maxFallSpeed)
209 {
210 rb.velocity = new Vector2(rb.velocity.x, -stats.maxFallSpeed);
211 }
